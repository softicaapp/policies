/// Firestore `app_config/muksang_notice` 문서 구조
/// {
/// "id": "notice_2026_03", ← 변경 시 일주일 안보기 무시하고 재노출
/// "enabled": true,
/// "storeUrl": "https://play.google.com/...", ← 없으면 버튼 미표시
/// "titleKo": "...", "titleEn": "...", "titleEs": "...", "titlePt": "...",
/// "bodyKo": "...", "bodyEn": "...", "bodyEs": "...", "bodyPt": "..."
/// }
class NoticeModel {
final String id;
final bool enabled;
final Map<String, String> titles;
final Map<String, String> bodies;
final String storeUrl;

const NoticeModel({
required this.id,
required this.enabled,
required this.titles,
required this.bodies,
required this.storeUrl,
});

factory NoticeModel.fromMap(Map<String, dynamic> map) {
return NoticeModel(
id: map['id'] as String? ?? '',
enabled: map['enabled'] as bool? ?? false,
storeUrl: map['storeUrl'] as String? ?? '',
titles: {
'ko': map['titleKo'] as String? ?? '',
'en': map['titleEn'] as String? ?? '',
'es': map['titleEs'] as String? ?? '',
'pt': map['titlePt'] as String? ?? '',
},
bodies: {
'ko': map['bodyKo'] as String? ?? '',
'en': map['bodyEn'] as String? ?? '',
'es': map['bodyEs'] as String? ?? '',
'pt': map['bodyPt'] as String? ?? '',
},
);
}

String title(String locale) => titles[locale] ?? titles['ko'] ?? '';
String body(String locale) => bodies[locale] ?? bodies['ko'] ?? '';
}

final noticeProvider = StreamProvider<NoticeModel?>((ref) {
return FirebaseFirestore.instance
.collection('app_config')
.doc('muksang_notice') // ← 앱마다 다른 문서명
.snapshots()
.map((snap) => snap.exists ? NoticeModel.fromMap(snap.data()!) : null);

class HomeScreen extends ConsumerStatefulWidget {
const HomeScreen({super.key});
@override
ConsumerState<HomeScreen> createState() => \_HomeScreenState();
}

class \_HomeScreenState extends ConsumerState<HomeScreen> {
bool \_noticeShown = false;

bool \_shouldShow(NoticeModel notice) {
if (!notice.enabled || notice.id.isEmpty) return false;
final box = Hive.box('settings');
final dismissedId = box.get('notice_dismissed_id', defaultValue: '');
final dismissedUntilStr = box.get('notice_dismissed_until', defaultValue: '');
if (dismissedId != notice.id) return true; // 새 공지 → 무조건 표시
if (dismissedUntilStr.isNotEmpty) {
final until = DateTime.tryParse(dismissedUntilStr);
if (until != null && DateTime.now().isBefore(until)) return false;
}
return true;
}

void _showNotice(NoticeModel notice) {
final str = ref.read(strProvider);
showDialog(
context: context,
barrierDismissible: true,
builder: (_) => \_NoticeDialog(notice: notice, str: str),
);
}

@override
Widget build(BuildContext context) {
ref.listen<AsyncValue<NoticeModel?>>(noticeProvider, (_, next) {
if (\_noticeShown) return;
next.whenData((notice) {
if (notice == null) return;
if (!\_shouldShow(notice)) return;
\_noticeShown = true;
WidgetsBinding.instance.addPostFrameCallback((_) => \_showNotice(notice));
});
});

class \_NoticeDialog extends StatelessWidget {

const \_NoticeDialog({required this.notice, required this.str});

@override

final cs = Theme.of(context).colorScheme;

title: Text(
notice.title(str.intlLocale),
style: GoogleFonts.notoSerif(fontWeight: FontWeight.bold, fontSize: 16),
),
content: Text(
notice.body(str.intlLocale),
style: GoogleFonts.notoSans(fontSize: 14, height: 1.6),

동작 규칙
enabled: false → 팝업 안 뜸
id 변경 → 일주일 안보기 무시하고 즉시 재노출
일주일 안보기 클릭 → 7일간 미표시
닫기 클릭 → 다음 실행 시 다시 표시
storeUrl 비어 있으면 "앱 보러가기" 버튼 미표시
앱 보러가기: 전체 너비 꽉 차는 ElevatedButton (가장 눈에 띄게)
일주일 안보기 / 닫기: 아래에 가운데 정렬된 TextButton 2개
